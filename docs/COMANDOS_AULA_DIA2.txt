import asyncio, os
from collections import Counter
from telethon import TelegramClient

async def main():
    c = TelegramClient(os.environ['TG_SESSION'], int(os.environ['TG_API_ID']),
                       os.environ['TG_API_HASH'])
    await c.connect()
    ext, menor, maior, ndoc = Counter(), None, 0, 0
    async for m in c.iter_messages(os.environ['CANAL'], limit=200):
        if m.document is None:
            continue                       # WebPage/foto nao e arquivo
        ndoc += 1
        f = m.file
        e = (f.ext or '?').lower().lstrip('.')
        sz = f.size or 0
        ext[e] += 1
        maior = max(maior, sz)
        if menor is None or sz < menor[0]:
            menor = (sz, f.name)
    print('documentos ... %d' % ndoc)
    print('extensoes .... %s' % dict(ext))
    print('maior ........ %.1f MB' % (maior / 1048576.0))
    if menor:
        print('menor ........ %.1f MB  %s' % (menor[0] / 1048576.0, menor[1]))
    await c.disconnect()

asyncio.run(main())
PY

import asyncio, os, re
from telethon import TelegramClient
RX = re.compile(r'(\d[\d\.]*)\s*(?:\(\d+\))?\s*\.txt$', re.I)
async def main():
    c = TelegramClient(os.environ['TG_SESSION'], int(os.environ['TG_API_ID']),
                       os.environ['TG_API_HASH'])
    await c.connect()
    razoes = []
    print('%-38s %10s %8s %9s' % ('ARQUIVO', 'BYTES', 'alegado', 'B/linha'))
    async for m in c.iter_messages(os.environ['CANAL'], limit=60):
        if m.document is None or not (m.file.name or '').lower().endswith('.txt'):
            continue
        mm = RX.search(m.file.name)
        if not mm:
            continue
        n = int(mm.group(1).replace('.', ''))
        r = m.file.size / float(n)
        razoes.append(r)
        print('%-38s %10d %8d %9.1f' % (m.file.name[:37], m.file.size, n, r))
    if razoes:
        print('min %.1f  max %.1f  media %.1f  (n=%d)'
              % (min(razoes), max(razoes), sum(razoes)/len(razoes), len(razoes)))
    await c.disconnect()
asyncio.run(main())

# Função auxiliar: resolve UTF-8 (acento) + evita ConvertTo-Json em JSON aninhado
function Es-Send {
  param(
    [string]$Method,          # Get, Post, Put, Delete
    [string]$Path,            # ex: "/apt_groups/_search?pretty"
    [string]$Json = $null     # o corpo, como TEXTO (here-string)
  )
  $args = @{ Method = $Method; Uri = "$ES$Path" }
  if ($Json) {
    $args.ContentType = "application/json; charset=utf-8"
    $args.Body        = [System.Text.Encoding]::UTF8.GetBytes($Json)
  }
  Invoke-RestMethod @args | ConvertTo-Json -Depth 20
}

  n=$(/usr/bin/curl -s -X POST "$ES/apt_groups/_count" -H 'Content-Type: application/json' \
      -d "{\"query\":{\"match_phrase\":{\"relatorio\":\"$termo\"}}}" \
